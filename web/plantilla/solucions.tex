\makeatletter
\newcommand{\departament}[1]{\def\@departament{#1}}
\newcommand{\trimestre}[1]{\def\@trimestre{#1}}

\newcommand{\crearcabecera}{%
  \begin{center}
    {\LARGE\bfseries Solucions\par}
    \vspace{0.5em}
    {\large\@departament%
      \ifx\@departament\@empty\else\ifx\@trimestre\@empty\else\ $\cdot$\ \fi\fi
      \@trimestre\par}
    \ifx\@unitat\@empty\else\vspace{0.2em}{\large\itshape\@unitat\par}\fi
  \end{center}
  \vspace{0.2em}\hrule\vspace{0.6em}}
\makeatother

% ---------------------------------------------------------
% Exercicis: \begin{exercici}[2,5] ... \end{exercici}
% (sense argument no es mostren els punts)
% ---------------------------------------------------------
\newcounter{numexercici}
\newenvironment{exercici}[1][]{%
  \refstepcounter{numexercici}%
  \setcounter{apartat}{0}\setcounter{solapartat}{0}%
  \par\vspace{1em}\noindent\textbf{Exercici \thenumexercici.}%
  \ifx&#1&\else\ \textit{(\textpunts{#1})}\fi
  \par\vspace{0.3em}\noindent\ignorespaces
}{%
  \par\vspace{0.5em}
}
% Procedència de l'exercici (opcional)
\newcommand{\procedencia}[1]{{\small\textit{(#1)}}\par\vspace{0.2em}\noindent\ignorespaces}
% Separació entre l'enunciat i la solució (quan s'inclouen els enunciats)
\newcommand{\titolsolucio}{\par\medskip\noindent\textbf{Solució}\par\nopagebreak\vspace{-0.3em}}
% Exercici sense solució (p. ex. un exercici propi on no se n'ha escrit cap)
\newcommand{\sensesolucio}{\textit{No s'ha escrit la solució d'aquest exercici.}\par}
