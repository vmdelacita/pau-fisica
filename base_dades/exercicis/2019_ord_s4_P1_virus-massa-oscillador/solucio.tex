\begin{solapartat}
\punts{0,4} $\omega = \sqrt{\dfrac{k}{m}}$

\punts{0,2} $\omega_\text{braç} = 2\pi f_\text{braç} = 2\pi\times 2,00\times 10^{15} = 4\pi\times 10^{15}\ \mathrm{rad\,s^{-1}}$

% DUBTE: a l'original falta el quadrat a (4π × 10^15); el resultat 3,32 × 10^13 correspon a (4π × 10^15)^2.
\punts{0,4} $k = m_\text{braç}\,\omega^2_\text{braç} = 2,10\times 10^{-19}\times\left(4\pi\times 10^{15}\right) = 3,32\times 10^{13}\ \mathrm{N\,m^{-1}}$
\end{solapartat}

\begin{solapartat}
\punts{0,2} $\omega_{\text{braç+virus}} = 2\pi f_{\text{braç+virus}} = 2\pi\times 2,87\times 10^{14} = 5,74\pi\times 10^{14}\ \mathrm{rad\,s^{-1}}$

\punts{0,3} $k = \left(m_\text{braç} + m_\mathrm{virus}\right)\omega^2_{\text{braç+virus}}$

\punts{0,4} $m_\text{braç} + m_\mathrm{virus} = \dfrac{k}{\omega^2_{\text{braç+virus}}} = \dfrac{3,32\times 10^{13}}{\left(5,74\pi\times 10^{14}\right)^2} = 1,02\times 10^{-17}\ \mathrm{kg}$

\punts{0,1}
\[ m_\mathrm{virus} = 1,02\times 10^{-17} - m_\text{braç} = 1,02\times 10^{-17} - 2,10\times 10^{-19} = 9,99\times 10^{-18}\ \mathrm{kg} \]
\end{solapartat}
