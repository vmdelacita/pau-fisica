\begin{solapartat}
\[ U = \frac{1}{2}kx^2 \]

\punts{0,2} El pendent de la recta equival a $\frac{1}{2}k$
\[ \mathit{Pendent} = \frac{\Delta y}{\Delta x} = \frac{\Delta U}{\Delta x^2} = \frac{200}{0,25} = 800\ \mathrm{J/m^2} \]

\punts{0,2} $\mathit{Pendent} = \frac{1}{2}k \Rightarrow k = 2\times \mathit{pendent} = 1600\ \mathrm{N/m}$

\punts{0,2} $T = 2\pi\sqrt{\dfrac{m}{k}}$
\[ T = 2\pi\sqrt{\frac{0,0625}{1600}} = 0,039\ \mathrm{s} \]
\figura[0.5]{sol_fig1.png}

\punts{0,4} $E_{c,\textit{màx}} = E = \frac{1}{2}kA^2$
\[ E_{c,\textit{màx}} = \frac{1}{2}1600\cdot (0,6)^2 = 288\ \mathrm{J} \]
\end{solapartat}

\begin{solapartat}
\punts{0,4} $\lambda = vT = 30\times 0,039 = 1,1781\ \mathrm{m}$
\begin{gather*}
\kappa = \frac{2\pi}{\lambda} = 5,33\ \mathrm{m^{-1}}\\
\omega = \frac{2\pi}{T} = \sqrt{\frac{k}{m}} = 160\ \mathrm{rad/s}
\end{gather*}

\punts{0,2} $y(x, t) = A\sin(\omega t - \kappa x)$\\
(també és vàlida l'expressió amb el cosinus i expressions equivalents de l'equació d'ona en funció de T, $\lambda$, \ldots)

\punts{0,4} $y(x, t) = 0,6\sin(160\,t - 5,33\,x)$, $x$ en m i $t$ en s.
\end{solapartat}
