\punts{0,2} $\phi = BA\cos\theta$

\smallskip
\punts{0,2} $A = \pi R^2 = \pi\times(0,05)^2 = 7,85\times 10^{-3}\,m^2$

\smallskip
\punts{0,2} $\phi_a = 60\times 10^{-3}\times 7,85\times 10^{-3}\times\cos 0° = 4,71\times 10^{-4}\,Wb$

\smallskip
\punts{0,2} $\phi_b = 60\times 10^{-3}\times 7,85\times 10^{-3}\times\cos 45° = 3,33\times 10^{-4}\,Wb$

\smallskip
\punts{0,2} $\phi_c = 60\times 10^{-3}\times 7,85\times 10^{-3}\times\cos 90° = 0$
\end{solapartat}

\begin{solapartat}
\punts{0,2} $\varepsilon = -\dfrac{d\phi}{dt} = NBA\omega\sin\theta$

\smallskip
\punts{0,2} $(a) \rightarrow$ l'angle entre $\vec{B}$ i $\vec{A}$ és $0 \Rightarrow \sin 0° = 0 \Rightarrow \varepsilon_a = 0$

\smallskip
\phantom{\punts{0,2}} En (b) i (c) l'angle entre $\vec{B}$ i $\vec{A}$ és $\neq 0 \Rightarrow \varepsilon_b \neq 0$ i $\varepsilon_c \neq 0$

\smallskip
\punts{0,2} $\omega = 300\,\dfrac{rev}{\mathrm{min}}\times\dfrac{2\pi rad}{1rev}\times\dfrac{1\,\mathrm{min}}{60s} = 10\pi\ \ rad\,s^{-1}$

\smallskip
\punts{0,2} $\varepsilon_b = 60\times 10^{-3}\times 7,85\times 10^{-3}\times 10\pi\times\sin 45° = 0,0105\,V$

\smallskip
\punts{0,2} $\varepsilon_c = 60\times 10^{-3}\times 7,85\times 10^{-3}\times 10\pi\times\sin 90° = 0,0148\,V$
\end{solapartat}
