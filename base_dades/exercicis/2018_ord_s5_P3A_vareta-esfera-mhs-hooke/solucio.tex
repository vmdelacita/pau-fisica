\medskip
\punts{0,3} $\left.\begin{aligned} \omega &= \sqrt{\dfrac{k}{m}}\\[2pt] T &= \dfrac{2\pi}{\omega} \end{aligned}\right\} \rightarrow T = 2\pi\sqrt{\dfrac{m}{k}} = 2\pi\sqrt{\dfrac{0,3}{200}} = 0,24\,s$
\end{solapartat}
