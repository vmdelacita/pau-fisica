\punts{0,3} $E = k\,\dfrac{q}{r^2}$
\begin{gather*}
E_\mathrm{Total} = E + 2E\sin 30 = E(1 + 2\sin 30)\\
\cos 30 = \frac{l/2}{r} \Rightarrow r = \frac{l/2}{\cos 30} = \frac{0,05}{\sqrt{3}/2} = 0,0577\ \mathrm{m}\\
E_\mathrm{Total} = k\,\frac{q}{r^2} + 2k\,\frac{q}{r^2}\,\frac{1}{2} = 2k\,\frac{q}{r^2}
\end{gather*}

\punts{0,4} $E_\mathrm{Total} = 2\times 8,99\times 10^{9}\,\dfrac{10^{-9}}{0,0577^2} = 5400\ \mathrm{N/C}$
