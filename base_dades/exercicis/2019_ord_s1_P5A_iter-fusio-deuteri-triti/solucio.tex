\begin{solapartat}
\punts{0,2} ${}^{2}_{1}H$, deuteri, 1 protó i 1 neutró

\smallskip
\punts{0,2} ${}^{3}_{1}H$, triti, 1 protó i 2 neutrons

\smallskip
\punts{0,2} ${}^{4}_{2}He$, heli, 2 protons i 2 neutrons

\smallskip
\punts{0,4} ${}^{2}_{1}H + {}^{3}_{1}H \rightarrow {}^{4}_{2}He + {}^{1}_{0}n$
\end{solapartat}

\begin{solapartat}
Disminució de la massa:

\smallskip
\punts{0,5} $\Delta m = \left[m({}^{2}_{1}H) + m({}^{3}_{1}H)\right] - \left[m({}^{4}_{2}He) + m({}^{1}_{0}n)\right] = 3,14\times 10^{-29}\ \mathrm{kg}$

\smallskip
\punts{0,5} $E = \Delta mc^2 = 3,14\times 10^{-29}\times (3,00\times 10^{8})^2 = 2,82\times 10^{-12}\ \mathrm{J}$
\end{solapartat}
