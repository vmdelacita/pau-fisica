\begin{solapartat}
\punts{0,25} $\omega = \sqrt{\dfrac{k}{m}}$

\punts{1}
\[
\left.
\begin{aligned}
\omega_\mathit{aranya+ins} &= \sqrt{\frac{k}{m_\mathit{aranya+ins}}}\\[4pt]
\omega_\mathit{aranya} &= \sqrt{\frac{k}{m_\mathit{aranya}}}
\end{aligned}
\right\}
\Rightarrow \frac{\omega_\mathit{aranya+ins}}{\omega_\mathit{aranya}} = \frac{f_\mathit{aranya+ins}}{f_\mathit{aranya}} = \sqrt{\frac{m_\mathit{aranya}}{m_\mathit{aranya+ins}}}
\]
\[
\frac{m_\mathit{aranya}}{m_\mathit{aranya+ins}} = \left(\frac{f_\mathit{aranya+ins}}{f_\mathit{aranya}}\right)^2
\Longrightarrow \frac{m_\mathit{aranya} + m_\mathit{ins}}{m_\mathit{aranya}} = \left(\frac{f_\mathit{aranya}}{f_\mathit{aranya+ins}}\right)^2 = \left(\frac{12}{10}\right)^2 = 1,44
\]
\[
1 + \frac{m_\mathit{ins}}{m_\mathit{aranya}} = 1,44 \Longrightarrow \frac{m_\mathit{ins}}{m_\mathit{aranya}} = 0,44 \Longrightarrow m_\mathit{aranya} = \frac{m_\mathit{ins}}{0,44} = 2,27\ \mathrm{g} = 2,27\times 10^{-3}\ \mathrm{kg}
\]
\end{solapartat}

\begin{solapartat}
\punts{0,1} $\omega = \sqrt{\dfrac{k}{m}} \Rightarrow k = m\omega^2$

\punts{0,65} $k = m\omega^2 = m_\mathit{aranya}\left(2\pi f_\mathit{aranya}\right)^2 = 2,27\times 10^{-3}\,(2\pi 12)^2 = 12,9\ \mathrm{N/m}$

\destaca{Alternativament:}

\punts{0,65} $k = m\omega^2 = m_\mathit{aranya+ins}\left(2\pi f_\mathit{aranya+ins}\right)^2 = 3,27\times 10^{-3}\,(2\pi 10)^2 = 12,9\ \mathrm{N/m}$

\punts{0,25} La màxima velocitat s'assoleix a la posició d'equilibri.

\punts{0,25} La màxima acceleració s'assoleix a les posicions de màxima amplitud.
\end{solapartat}
