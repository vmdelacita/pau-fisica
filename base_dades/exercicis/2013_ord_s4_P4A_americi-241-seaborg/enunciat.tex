\begin{gather*}
{}^{240}_{\phantom{2}94}\mathrm{Pu} + {}^{a}_{b}\mathrm{x} \rightarrow {}^{241}_{\phantom{2}94}\mathrm{Pu}\\
{}^{241}_{\phantom{2}94}\mathrm{Pu} \rightarrow {}^{241}_{\phantom{2}95}\mathrm{Am} + {}^{c}_{d}\mathrm{Y}
\end{gather*}
