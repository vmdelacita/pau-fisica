\begin{solapartat}
\punts{0,25} $\eta = 0,9$, llavors la potència al primari, $P_1$, ha de ser: $P_1 = \dfrac{P_2}{\eta} = \dfrac{18}{0,9} = 20\ \mathrm{kW}$

\punts{0,5} La potència ve donada per: $P_1 = V_1 I_1$

\punts{0,25} Llavors al primari $I_1 = \dfrac{P_1}{V_1} = \dfrac{20.000\ \mathrm{W}}{25.000\ \mathrm{V}} = 0,8\ \mathrm{A}$

\punts{0,25} i al secundari $I_2 = \dfrac{P_2}{V_2} = \dfrac{18.000\ \mathrm{W}}{220\ \mathrm{V}} = 81,8\ \mathrm{A}$
\end{solapartat}
