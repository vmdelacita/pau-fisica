\begin{solapartat}
\punts{0,2} L'energia del fotó serà igual a la disminució de la massa per la velocitat de la llum al quadrat.\\
$E = \Delta\mathrm{m}c^2$

\punts{0,2} $\Delta m = [m({}^{1}_{1}p) + m({}^{1}_{0}n)] - m({}^{2}_{1}H)$

\punts{0,1} $\Delta m = 3,96\times 10^{-30}\ \mathrm{kg}$

\punts{0,1} $E = \Delta mc^2 = 3,96\times 10^{-30}\times (3,00\times 10^{8})^2 = 3,564\times 10^{-13}\ \mathrm{J}$

\punts{0,1} $E = 3,564\times 10^{-13}\ \mathrm{J}\cdot\dfrac{1\ \mathrm{eV}}{1,602\times 10^{-19}\ \mathrm{J}} = 2,22\times 10^{6}\ \mathrm{eV}$

\punts{0,3} $f = \dfrac{E}{h} = \dfrac{3,564\times 10^{-13}}{6,63\times 10^{-34}} = 5,38\times 10^{20}\ \mathrm{Hz}$
\end{solapartat}
