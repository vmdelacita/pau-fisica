\begin{itemize}
  \item[---] L'ona es reflecteix en una superfície. \hfill\textit{[0,3 punts]}
  \item[---] L'ona passa de l'aire a l'aigua (on la velocitat del so és més gran). \hfill\textit{[0,3 punts]}
  \item[---] El focus sonor es posa en moviment en direcció a nosaltres. \hfill\textit{[0,4 punts]}
\end{itemize}
\end{apartat}
