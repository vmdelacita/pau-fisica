\begin{align*}
&E_m = 0 = \frac{1}{2}\,m\,v^2 - G\,\frac{M_C m}{R_C} \quad \text{\punts{0,2}}\\
&\Rightarrow v = \sqrt{\frac{2\,G\,M_C}{R_C}} = \sqrt{\frac{2\cdot 6,67\cdot 10^{-11}\,9,43\cdot 10^{20}}{477\cdot 10^{3}}} = 514\ \mathrm{m/s} \quad \text{\punts{0,2}}
\end{align*}
\end{solapartat}
