\begin{solapartat}
\[ G\frac{M_L}{(R_L+h)^2} = (R_L+h)\,\omega^2 \ \text{\punts{0,4}} \Rightarrow M_L = \frac{(R_L+h)^3}{G}\,\omega^2 \ \text{\punts{0,2}} \]
\[ M_L = \frac{\left(1,74\times 10^{6} + 10^{5}\right)^3}{6,67\times 10^{-11}}\left(\frac{2\pi}{1,18\times 10^{2}\cdot 60}\right)^2 = 7,36\times 10^{22}\ \mathrm{kg} \ \text{\punts{0,2}} \]
\[ g_L = G\frac{M_L}{R_L^2} = 1,62\ \mathrm{m/s^2} \ \text{\punts{0,2}} \]
\end{solapartat}

\begin{solapartat}
\[ E_{\textit{mecànica superfície Lluna}} = -G\,\frac{M_L\,m}{R_L} + \frac{1}{2}m\,v^2 \ \text{\punts{0,5}} \]
Un objecte es podrà escapar de la superfície de la Lluna si la seva energia mecànica és zero \punts{0,25} $\Rightarrow$
\begin{align*}
&-G\,\frac{M_L\,\not{m}}{R_L} + \frac{1}{2}\,\not{m}\,v^2 = 0 \Rightarrow v = \sqrt{\frac{2G\,M_L}{R_L}}\\
&\quad = \sqrt{\frac{2\times 6,67\times 10^{-11}\ 7,36\times 10^{22}}{1,74\times 10^{6}}} = 2,38\times 10^{3}\ \mathrm{m/s} \ \text{\punts{0,25}}
\end{align*}
\end{solapartat}
